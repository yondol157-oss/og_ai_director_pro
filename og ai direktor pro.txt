const fileToBase64 = (file) => {
  return new Promise((resolve, reject) => {
    if (file.size > 20 * 1024 * 1024) {
      reject(new Error("Ukuran berkas maksimal 20MB"));
      return;
    }
    const reader = new FileReader();
    reader.onload = () => {
      const result = reader.result;
      const mimeType = result.split(';')[0].split(':')[1];
      const data = result.split(',')[1];
      resolve({ mimeType, data, previewUrl: result });
    };
    reader.onerror = error => reject(error);
    reader.readAsDataURL(file);
  });
};

const downloadImage = async (dataUrl, filename) => {
  try {
    const response = await fetch(dataUrl);
    const blob = await response.blob();
    const blobUrl = window.URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.style.display = 'none';
    link.href = blobUrl;
    link.download = filename || `og-ai-director-${Date.now()}.png`;
    document.body.appendChild(link);
    link.click();
    setTimeout(() => {
      document.body.removeChild(link);
      window.URL.revokeObjectURL(blobUrl);
    }, 120);
  } catch (err) {
    console.error("Gagal mengunduh gambar:", err);
  }
};

const copyToClipboard = (text) => {
  try {
    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.style.position = 'fixed'; 
    textarea.style.opacity = '0';
    document.body.appendChild(textarea);
    textarea.select();
    const success = document.execCommand('copy');
    document.body.removeChild(textarea);
    return success;
  } catch (err) {
    console.error("Gagal menyalin draf:", err);
    return false;
  }
};

const callGeminiText = async (prompt, imageInputs = []) => {
  const apiKey = getApiKey();
  const url = `https://generativelanguage.googleapis.com/v1beta/models/${AI_CONFIG.TEXT_MODEL}:generateContent?key=${apiKey}`;
  
  const strictSystemInstruction = "CRITICAL INSTRUCTION: Output directly the requested script, concepts, or JSON without any conversational filler, introductory remarks, greetings, or self-referential titles. Start directly with the content. Ensure 100% strict relevance and absolute adherence to the provided topic/niche.";
  const combinedPrompt = `${strictSystemInstruction}\n\n${prompt}`;

  const parts = [{ text: combinedPrompt }];
  imageInputs.forEach((img) => {
    if (img && img.data) {
      parts.push({
        inlineData: { mimeType: img.mimeType, data: img.data }
      });
    }
  });

  const payload = {
    contents: [{ parts: parts }],
    generationConfig: prompt.includes("JSON_OUTPUT_REQUIRED") ? { responseMimeType: "application/json" } : {}
  };

  const response = await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload)
  });

  if (!response.ok) throw new Error(`Gemini Text Error: ${response.status}`);
  const data = await response.json();
  return data.candidates?.[0]?.content?.parts?.[0]?.text || "Maaf, respon teks belum dapat digenerate.";
};

const callGeminiImage = async (prompt, imageInputs = [], aspectRatio = "1:1") => {
  const apiKey = getApiKey();
  const url = `https://generativelanguage.googleapis.com/v1beta/models/${AI_CONFIG.IMAGE_MODEL}:generateContent?key=${apiKey}`;
  
  const enhancedPrompt = `${prompt}\n\n[100% LOCKED ABSOLUTE VISUAL INSTRUCTION]: Render this exact prompt composition with uncompromising visual fidelity, exact matching clothing color/outfit style from reference, and absolute adherence to every single detail. Aspect ratio: ${aspectRatio}. Photorealistic textures, calibrated dynamic range, exact visual synchronization with the prompt description.`;
  
  const parts = [{ text: enhancedPrompt }];
  imageInputs.forEach(img => {
    if (img && img.data) {
      parts.push({ inlineData: { mimeType: img.mimeType, data: img.data } });
    }
  });

  const payload = {
    contents: [{ parts: parts }],
    generationConfig: { 
      responseModalities: ["TEXT", "IMAGE"],
      imageConfig: { aspectRatio: aspectRatio }
    }
  };

  const response = await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(payload)
  });

  if (!response.ok) throw new Error(`Gemini Image Error: ${response.status}`);
  const data = await response.json();
  
  const candidateParts = data.candidates?.[0]?.content?.parts || [];
  
  for (const p of candidateParts) {
    if (p.inlineData && p.inlineData.data) {
      return `data:${p.inlineData.mimeType || 'image/png'};base64,${p.inlineData.data}`;
    }
    if (p.fileData && p.fileData.fileUri) {
      return p.fileData.fileUri;
    }
    if (p.text && p.text.startsWith('data:image')) {
      return p.text;
    }
  }

  // Fallback: Check if there is an image in any part object structure
  for (const candidate of (data.candidates || [])) {
    for (const part of (candidate.content?.parts || [])) {
      if (part.inline_data && part.inline_data.data) {
        return `data:${part.inline_data.mime_type || 'image/png'};base64,${part.inline_data.data}`;
      }
    }
  }

  throw new Error("Model belum berhasil mengembalikan representasi visual.");
};

const parseJsonSafe = (rawText, fallbackValue = []) => {
  try {
    let clean = rawText.replace(/```json/g, '').replace(/```/g, '').trim();
    let firstB = clean.indexOf('[');
    let lastB = clean.lastIndexOf(']');
    if (firstB !== -1 && lastB !== -1 && lastB > firstB) {
      clean = clean.substring(firstB, lastB + 1);
    }
    return JSON.parse(clean);
  } catch (err) {
    console.warn("JSON parse fallback triggered:", err);
    return fallbackValue;
  }
};

const CharacterSheetStudio = () => {
  const [refImage, setRefImage] = useState(null);
  const [characterName, setCharacterName] = useState('');
  const [style, setStyle] = useState(OPTIONS.imageStyles[0]);
  const [loading, setLoading] = useState(false);
  const [loadingMsg, setLoadingMsg] = useState('');
  const [sheets, setSheets] = useState({
    portrait: null,
    closeup: null,
    fullBody: null,
    turnaround: null
  });

  const handleGenerateSheets = async () => {
    if (!refImage) return;
    setLoading(true);
    setSheets({ portrait: null, closeup: null, fullBody: null, turnaround: null });

    try {
      const charTag = characterName.trim() ? `character named "${characterName.trim()}"` : "the character";

      setLoadingMsg("Merender Character Portrait Sheet berlatar putih bersih dengan pakaian & muka yang konsisten 100%...");
      const portraitPrompt = `100% LOCKED CHARACTER PORTRAIT SHEET: Create a professional character headshot and facial expression emotion sheet for ${charTag} based strictly on the provided reference image. Absolute uncompromising consistency: match exact face, facial features, exact eye shape, nose, hairstyle, and exact outfit (matching color, fabric, and style) from reference photo. Artistic style: ${style}. Multiple head angles showing detailed facial features and clean expressions. SOLID PURE WHITE BACKGROUND (#FFFFFF), studio lighting, high resolution.`;
      const portraitImg = await callGeminiImage(portraitPrompt, [refImage], "1:1");
      setSheets(prev => ({ ...prev, portrait: portraitImg }));

      setLoadingMsg("Merender Close-Up Facial Portrait berlatar putih bersih dengan wajah & baju identik...");
      const closeupPrompt = `100% LOCKED CLOSE-UP FACIAL PORTRAIT: Create a high-detail close-up facial portrait framing for ${charTag} based strictly on the provided reference image. Absolute uncompromising consistency: exact identical face, exact facial features, eyes, nose, lips, hair, and exact identical clothing style/color from reference photo. Artistic style: ${style}. Sharp facial focus, expressive eyes. SOLID PURE WHITE BACKGROUND (#FFFFFF), studio portrait lighting, 8k resolution.`;
      const closeupImg = await callGeminiImage(closeupPrompt, [refImage], "1:1");
      setSheets(prev => ({ ...prev, closeup: closeupImg }));

      setLoadingMsg("Merender Full Body Character Reference Sheet berlatar putih bersih...");
      const fullBodyPrompt = `100% LOCKED FULL BODY REFERENCE SHEET: Create a complete full body character reference sheet for ${charTag} based strictly on the provided reference image. Absolute uncompromising consistency: exact face, facial features, hair, and exact identical clothing color/style from reference photo. Artistic style: ${style}. Full standing pose displaying full outfit and distinct proportions. SOLID PURE WHITE BACKGROUND (#FFFFFF), professional studio character design sheet layout, 8k resolution.`;
      const fullBodyImg = await callGeminiImage(fullBodyPrompt, [refImage], "3:4");
      setSheets(prev => ({ ...prev, fullBody: fullBodyImg }));

      setLoadingMsg("Merender Multi-View Character Sheet / Turnaround berlatar putih bersih...");
      const turnaroundPrompt = `100% LOCKED MULTI-VIEW CHARACTER TURNAROUND: Create a professional multi-view character turnaround sheet showing front view, 3/4 view, side profile view, and back view for ${charTag} based strictly on the provided reference image. SOLID PURE WHITE BACKGROUND (#FFFFFF), technical character model sheet, exact consistent proportions across all angles.`;
      const turnaroundImg = await callGeminiImage(turnaroundPrompt, [refImage], "16:9");
      setSheets(prev => ({ ...prev, turnaround: turnaroundImg }));

    } catch (err) {
      console.error("Character Sheet generation error:", err);
    } finally {
      setLoading(false);
      setLoadingMsg('');
    }
  };

  return (
    <div className="space-y-8 animate-in max-w-5xl mx-auto">
      {loading && <GeneratingOverlay message={loadingMsg || "Memproses aset Character Sheet konsistensi tinggi secara bersamaan..."} />}

      <div className="p-6 rounded-3xl bg-[#0a0f1d]/70 border border-cyan-950/90 shadow-xl backdrop-blur-md space-y-6">
        <div className="flex items-center gap-3 border-b border-cyan-950 pb-4">
          <div className="p-2.5 rounded-xl bg-cyan-500/10 text-cyan-400 border border-cyan-500/20">
            <Users size={22} />
          </div>
          <div>
            <h3 className="text-base font-black text-cyan-200 uppercase tracking-wider">Character Sheet Studio (Konsistensi Video 100%)</h3>
            <p className="text-xs text-gray-400">Unggah 1 foto referensi untuk menghasilkan acuan konsistensi karakter (termasuk close-up wajah & baju sama persis) berlatar putih bersih secara instan.</p>
          </div>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
          <div>
            <span className="text-xs font-bold text-gray-300 block mb-2 uppercase tracking-wider">Foto Referensi Utama (Wajib)</span>
            <ImageUploadSlot label="" value={refImage} onUpload={setRefImage} onRemove={() => setRefImage(null)} compact={false} />
          </div>

          <div className="space-y-4 flex flex-col justify-between">
            <div className="space-y-4">
              <FormGroup label="Nama Karakter (Opsional)">
                <Input placeholder="Contoh: Alex, Maya, Detektif Budi" value={characterName} onChange={e => setCharacterName(e.target.value)} />
              </FormGroup>

              <FormGroup label="Gaya Visual / Seni">
                <Select options={OPTIONS.imageStyles} value={style} onChange={e => setStyle(e.target.value)} />
              </FormGroup>
            </div>

              <Sparkles size={18} /> GENERATE CHARACTER SHEETS (LATAR PUTIH)
            </Button>
          </div>
        </div>
      </div>

      {(sheets.portrait || sheets.closeup || sheets.fullBody || sheets.turnaround) && (
        <div className="space-y-6 animate-in zoom-in">
          <div className="p-5 rounded-3xl bg-[#0a0f1d] border border-emerald-800/60 shadow-2xl space-y-6">

    setGeneratedScriptResult('');
    setRenderedShotsList([]);
    
    try {
      setProgressMsg("Direktur AI menganalisis premis & merumuskan naskah shot panel...");
      
      const charRefName = useSavedChar && selectedCharId 
        ? savedCharacters.find(c => String(c.id) === String(selectedCharId))?.name

        : "Tanpa dialog bicara karakter (fokus visual murni).";

      const promptPlanning = `
JSON_OUTPUT_REQUIRED
Sebagai sutradara sinematografi, susun naskah storyboard untuk cerita berikut secara SANGAT RELEVAN dan berfokus mutlak pada topik: "${storyDesc}"
Premis: ${storyDesc}
Jumlah Shot: ${numShots} (Setiap shot berdurasi TEPAT 10 DETIK padat isi)
Karakter: ${charRefName}
${dialogueInstruction}

[CRITICAL WARDROBE & FACE LOCK]: Every single shot in this storyboard MUST maintain 100% strict absolute consistency: the exact same character, exact same clothing, exact same outfit colors, exact same hairstyle, and exact same facial features across all shots. The wardrobe must not change.

Berikan respon JSON MURNI tanpa backtick dengan format persis:
[
  { "id": 1, "action": "Aksi utama talent", "dialogue": "Dialog karakter (jika ada/dibuat otomatis)", "cameraAngle": "Angle kamera", "description": "Deskripsi visual adegan untuk generator gambar AI dengan menjaga konsistensi mutlak baju, wajah, dan subjek dari referensi" }
]
      `.trim();

      const textRes = await callGeminiText(promptPlanning);

        { id: 1, action: "Aksi awal pembuka", dialogue: includeDialogue ? "Dialog otomatis yang relevan" : "", cameraAngle: "Eye Level", description: "Visual sinematik pembuka konsisten" }
      ]).slice(0, parseInt(numShots, 10));

      setGeneratedScriptResult(scriptSummaryText);

      const inputs = [];
      if (useSavedChar && selectedCharId) {
        const charObj = savedCharacters.find(c => String(c.id) === String(selectedCharId));
        if (charObj && charObj.image) {
          const mime = charObj.image.split(';')[0].split(':')[1];
          const data = charObj.image.split(',')[1];
          inputs.push({ mimeType: mime, data });
        }
      }
      images.char.forEach(img => img && inputs.push(img));
      images.outfit.forEach(img => img && inputs.push(img));
      images.loc.forEach(img => img && inputs.push(img));

      const renderedShots = [];
      for (let i = 0; i < parsed.length; i++) {
        const s = parsed[i];
        setProgressMsg(`Merender visual shot ${i + 1}: ${s.action.substring(0, 30)}...`);
        const shotPrompt = `[ABSOLUTE 100% WARDROBE, OUTFIT COLOR & FACE LOCK]: You MUST clone the EXACT clothing style, exact outfit colors, fabric, and facial features from the provided reference images. DO NOT alter the outfit, colors, or appearance across shots.\n\n[SCENE DETAILS]\n- Story Context: "${storyDesc}"\n- Scene Action: ${s.action}\n- Visual Description: "${s.description}"\n- Camera Angle: ${s.cameraAngle}\n- Art Style: ${style}\n\n[CRITICAL]: The character's clothes, colors, and face must remain 100% identical and synchronized across all panels. Professional cinematic composition, no text.`.trim();

      const cellTotalHeight = cellImgHeight + cellTextHeight;

        ctx.font = 'bold 20px "Segoe UI", Roboto, Helvetica, sans-serif';

        ctx.font = 'bold 16px "Segoe UI", Roboto, Helvetica, sans-serif';
        ctx.fillText(`[ ${item.cameraAngle.toUpperCase()} ]`, x + 130, y + cellImgHeight + 29);

        let currentY = y + cellImgHeight + 85;
        const wrapText = (text, startX, startY, maxWidth, lineHeight, fontStyle, color) => {
            ctx.fillStyle = color;
            ctx.font = fontStyle;
            const words = text.split(' ');
            let line = '';
            for(let n = 0; n < words.length; n++) {
                let testLine = line + words[n] + ' ';
                if (ctx.measureText(testLine).width > maxWidth && n > 0) {
                    ctx.fillText(line, startX, startY);
                    line = words[n] + ' ';
                    startY += lineHeight;
                } else {
                    line = testLine;
                }
            }
            ctx.fillText(line, startX, startY);
            return startY + lineHeight;
        };

    const unsub = onAuthStateChanged(auth, setUser);
    return () => unsub();
  }, []);

  useEffect(() => {
    if (!firebaseReady || !user) return;
    const writeLog = async () => {
      try {
        const email = user.email || `director-${user.uid.substring(0, 5)}@ai.com`;
        const logRef = doc(db, 'artifacts', appId, 'public', 'data', 'users_log', user.uid);
        await setDoc(logRef, {
          uid: user.uid,
          email,
          lastUsed: new Date().toISOString(),
          type: 'Google AI Pro Account'
        }, { merge: true });
      } catch (e) {
        console.warn("Log write error:", e);
      }
    };
    writeLog();
  }, [user]);

  useEffect(() => {
    if (!firebaseReady || !user) return;
    const logCol = collection(db, 'artifacts', appId, 'public', 'data', 'users_log');
    const unsub = onSnapshot(logCol, (snapshot) => {
      const logs = [];
      snapshot.forEach(d => logs.push(d.data()));
      logs.sort((a, b) => new Date(b.lastUsed) - new Date(a.lastUsed));
      setUserLogs(logs);
    }, err => console.warn(err));
    return () => unsub();
  }, [user]);

  if (!hasEntered) {
    return (
      <div 
        className="flex flex-col items-center justify-center min-h-screen w-full bg-[#070b14] relative overflow-hidden px-4"
        style={{ backgroundImage: `url("${TECH_GRID_SVG}")`, backgroundSize: '60px' }}
      >
        <div className="absolute inset-0 bg-gradient-to-t from-[#050811] via-transparent to-[#050811]/70 pointer-events-none" />
        <div className="absolute -top-40 -left-40 w-96 h-96 bg-cyan-500/10 rounded-full blur-3xl pointer-events-none" />
        <div className="absolute -bottom-40 -right-40 w-96 h-96 bg-amber-500/10 rounded-full blur-3xl pointer-events-none" />

        <div className="relative z-10 text-center space-y-8 max-w-2xl w-full p-8 sm:p-12 rounded-3xl bg-[#0a0f1d]/80 border border-cyan-800/40 backdrop-blur-xl shadow-2xl shadow-cyan-950/50">
          <div className="inline-flex p-4 rounded-2xl bg-gradient-to-br from-cyan-500 to-teal-400 shadow-xl shadow-cyan-500/25 text-[#050b14]">
            <Clapperboard size={48} strokeWidth={2.5} />
          </div>

          <div className="space-y-3">
            <span className="text-[11px] font-bold text-amber-400 tracking-widest uppercase px-3 py-1 rounded-full bg-amber-500/10 border border-amber-500/30">
              NEXT-GEN AI PRODUCTION ENVIRONMENT
            </span>
            <h1 className="text-3xl sm:text-5xl font-black tracking-tight text-white leading-tight font-sans">
              OG AI <br />
              <span className="bg-gradient-to-r from-cyan-400 via-teal-300 to-amber-300 bg-clip-text text-transparent">
                DIRECTOR PRO
              </span>
            </h1>
            <p className="text-gray-400 text-xs sm:text-sm max-w-md mx-auto leading-relaxed">
              Studio orkestrasi konten profesional berbasis AI. Hasilkan naskah afiliasi konversi tinggi, prompt sinematik, first frame photorealistic, dan visual storyboard otomatis.
            </p>
          </div>

          <div className="pt-2">
            <button 
              onClick={() => setHasEntered(true)}
              className="inline-flex items-center gap-3 px-8 py-4 bg-gradient-to-r from-cyan-500 via-teal-400 to-emerald-400 hover:from-cyan-400 hover:to-teal-300 text-[#050b14] font-black text-base rounded-2xl shadow-xl shadow-cyan-500/30 transform active:scale-95 transition-all cursor-pointer"
            >
              MASUK KE STUDIO SUTRADARA <ArrowRight size={20} />
            </button>
          </div>
        </div>
      </div>
    );
  }

  const ActiveComponent = TABS.find(t => t.id === activeTab)?.component || TABS[0].component;

  return (
    <div 
      className="flex h-screen w-full bg-[#070b14] text-gray-100 font-sans overflow-hidden selection:bg-cyan-500/30"
      style={{ backgroundImage: `url("${TECH_GRID_SVG}")`, backgroundSize: '60px' }}
    >
      {sidebarOpen && (
        <div 
          className="fixed inset-0 bg-black/80 z-40 md:hidden backdrop-blur-sm"
          onClick={() => setSidebarOpen(false)}
        />
      )}

      <aside 
        className={`fixed md:relative z-50 h-full bg-[#0a0f1d]/90 border-r border-cyan-950/80 transition-all duration-300 flex flex-col backdrop-blur-xl
          ${sidebarOpen ? 'translate-x-0' : '-translate-x-full md:translate-x-0'}
          ${sidebarCollapsed ? 'w-20' : 'w-72'}
        `}
      >
        <div className="p-5 flex items-center justify-between border-b border-cyan-950 h-20 shrink-0">
          {!sidebarCollapsed ? (
            <div className="flex items-center gap-3 overflow-hidden">
              <div className="w-9 h-9 rounded-xl bg-gradient-to-br from-cyan-500 to-teal-400 flex items-center justify-center shrink-0 text-[#050b14] shadow-md shadow-cyan-500/30">
                <Clapperboard size={20} strokeWidth={2.5} />
              </div>
              <div className="flex flex-col min-w-0">
                <span className="font-black text-sm tracking-tight text-white truncate">OG AI</span>
                <span className="text-[10px] text-cyan-400 font-bold tracking-wider uppercase">Director Pro</span>
              </div>
            </div>
          ) : (
            <div className="w-9 h-9 mx-auto rounded-xl bg-gradient-to-br from-cyan-500 to-teal-400 flex items-center justify-center text-[#050b14]">
              <Clapperboard size={20} strokeWidth={2.5} />
            </div>
          )}
          <button className="md:hidden text-gray-400 hover:text-white" onClick={() => setSidebarOpen(false)}>
            <X size={22} />
          </button>
        </div>

        <nav className="flex-1 overflow-y-auto p-3 space-y-1.5 custom-scrollbar">
          {TABS.map((tab) => {
            const Icon = tab.icon;
            const isActive = activeTab === tab.id;
            return (
              <button
                key={tab.id}
                onClick={() => { setActiveTab(tab.id); setSidebarOpen(false); }}
                className={`relative flex items-center gap-3.5 px-3.5 py-3 rounded-xl transition-all w-full text-left group cursor-pointer
                  ${isActive 
                    ? 'bg-cyan-500/10 text-white font-extrabold border border-cyan-500/30 shadow-inner' 
                    : 'text-gray-400 hover:bg-[#0f172a]/60 hover:text-cyan-200 border border-transparent'}
                  ${sidebarCollapsed ? 'justify-center px-0' : ''}
                `}
                title={sidebarCollapsed ? tab.title : ''}
              >
                {isActive && (
                  <div className="absolute left-0 top-1/2 -translate-y-1/2 w-1.5 h-7 bg-cyan-400 rounded-r-full shadow-md shadow-cyan-400" />
                )}
                <Icon size={20} className={`shrink-0 ${isActive ? 'text-cyan-400' : 'group-hover:text-cyan-300'}`} />
                {!sidebarCollapsed && (
                  <div className="flex flex-col min-w-0">
                    <span className="text-xs font-bold truncate">{tab.title}</span>
                    <span className="text-[10px] text-gray-500 uppercase tracking-wider">{tab.subtitle}</span>
                  </div>
                )}
              </button>
            );
          })}
        </nav>

        <div className="p-4 border-t border-cyan-950 hidden md:flex justify-center">
          <button 
            onClick={() => setSidebarCollapsed(!sidebarCollapsed)}
            className="p-2 bg-[#0d1424] text-gray-400 hover:text-cyan-300 rounded-xl border border-cyan-950 transition-colors cursor-pointer"
          >
            {sidebarCollapsed ? <ChevronRight size={18} /> : <ChevronLeft size={18} />}
          </button>
        </div>
      </aside>

      <main className="flex-1 flex flex-col h-full min-w-0 overflow-hidden relative">
        <header className="md:hidden flex items-center justify-between p-4 border-b border-cyan-950 bg-[#0a0f1d]/90 backdrop-blur-md sticky top-0 z-30">
          <button onClick={() => setSidebarOpen(true)} className="p-2 text-cyan-400 bg-[#0d1424] rounded-xl border border-cyan-950">
            <Menu size={20} />
          </button>
          <div className="font-black text-xs text-white uppercase tracking-wider flex items-center gap-1.5">
            <span className="text-cyan-400">OG AI</span> • {TABS.find(t => t.id === activeTab)?.title}
          </div>
          <div className="w-8"></div>
        </header>

        <header className="hidden md:flex h-20 items-center justify-between px-8 border-b border-cyan-950/80 bg-[#0a0f1d]/60 backdrop-blur-md shrink-0">
          <div>
            <h2 className="text-xl font-black text-white flex items-center gap-3">
              {TABS.find(t => t.id === activeTab)?.title}
              <span className="text-[11px] px-2.5 py-0.5 rounded-full bg-cyan-500/10 text-cyan-300 border border-cyan-500/30 uppercase tracking-widest font-mono font-bold">
                {TABS.find(t => t.id === activeTab)?.subtitle}
              </span>
            </h2>
            <p className="text-xs text-gray-400 mt-0.5">Konfigurasi sutradara kreatif dan eksekusi rendering visual instan.</p>
          </div>
          <div className="flex items-center gap-2"></div>
        </header>

        <div className="flex-1 overflow-y-auto p-4 md:p-8 custom-scrollbar">
          <div className="max-w-7xl mx-auto pb-16">
            <ActiveComponent />
          </div>
        </div>
      </main>

      <style dangerouslySetInnerHTML={{__html: `
        .custom-scrollbar::-webkit-scrollbar { width: 6px; height: 6px; }
        .custom-scrollbar::-webkit-scrollbar-track { background: transparent; }
        .custom-scrollbar::-webkit-scrollbar-thumb { background: #164e63; border-radius: 9999px; }
        .custom-scrollbar::-webkit-scrollbar-thumb:hover { background: #0891b2; }

        @keyframes fadeIn { from { opacity: 0; transform: translateY(6px); } to { opacity: 1; transform: translateY(0); } }
        @keyframes zoomIn { from { opacity: 0; transform: scale(0.97); } to { opacity: 1; transform: scale(1); } }
        
        .animate-in { animation-duration: 0.3s; animation-fill-mode: both; animation-timing-function: cubic-bezier(0.16, 1, 0.3, 1); }
        .fade-in { animation-name: fadeIn; }
        .zoom-in { animation-name: zoomIn; }
      `}} />
    </div>
  );
}
